cmake_minimum_required(VERSION 3.16)
project(HW3 C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

find_package(Threads REQUIRED)
set(hotel_targets hotel)
foreach(role manager administrator client)
    if(role STREQUAL "administrator")
        set(program hotel_admin)
    else()
        set(program hotel_${role})
    endif()
    add_executable(${program} independent/${role}.c independent/hotel.c)
    target_link_libraries(${program} PRIVATE Threads::Threads)
    target_compile_options(${program} PRIVATE -Wall -Wextra -Wpedantic)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        target_link_libraries(${program} PRIVATE rt)
    endif()
endforeach()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    list(APPEND hotel_targets hotel_unnamed)
else()
    message(STATUS "hotel_unnamed доступен только в Linux")
endif()
foreach(target IN LISTS hotel_targets)
    add_executable(${target} ${target}.c)
    target_link_libraries(${target} PRIVATE Threads::Threads)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        target_link_libraries(${target} PRIVATE rt)
    endif()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
